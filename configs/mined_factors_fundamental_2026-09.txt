# profitability
RANK($roa)
RANK($roe)
RANK($operating_margin)

# value: book and earnings against price
RANK($book_per_share / ($close + 1e-8))
RANK($earnings_per_share / ($close + 1e-8))

# quality and balance-sheet risk
-RANK($leverage)
RANK($asset_turnover)
-RANK($accrual_gap)

# change-based: fundamentals move slowly, so the DELTA carries the news
RANK(TS_DELTA($roa, 63))
RANK(TS_DELTA($operating_margin, 63))

# interactions with price behaviour
RANK($roa) - RANK(TS_STD($return, 21))
RANK($book_per_share / ($close + 1e-8)) - RANK(TS_MEAN($return, 63))
